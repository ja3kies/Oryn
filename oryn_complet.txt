private val routeOrder = listOf("home", "explore", "bookmarks", "settings")

private val importMimeTypes = arrayOf(
    "application/epub+zip",
    "application/zip",
    "application/pdf",
    "text/plain",
    "application/octet-stream"
)

class MainActivity : ComponentActivity() {

    private val viewModel: MainViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            val themeMode by viewModel.themeMode.collectAsState()
            val accentTheme by viewModel.accentTheme.collectAsState()
            val systemDarkMode = isSystemInDarkTheme()
            val motionEnabled = systemAnimationsEnabled(this@MainActivity)
            val darkMode = when (themeMode) {
                AppThemeMode.SYSTEM -> systemDarkMode
                AppThemeMode.LIGHT -> false
                AppThemeMode.DARK -> true
            }

            var route by rememberSaveable { mutableStateOf("home") }
            var selectedBook by remember { mutableStateOf<BookEntity?>(null) }

            val booksState by viewModel.books.collectAsState()
            val books = booksState.orEmpty()
            val favorites by viewModel.favoriteBooks.collectAsState()

            val importLauncher = rememberLauncherForActivityResult(
                ActivityResultContracts.OpenMultipleDocuments()
            ) { uris -> viewModel.importBooks(uris) }

            val importFavoriteLauncher = rememberLauncherForActivityResult(
                ActivityResultContracts.OpenMultipleDocuments()
            ) { uris -> viewModel.importBooks(uris, favorite = true) }

            OrynTheme(
                darkTheme = darkMode,
                accentTheme = accentTheme
            ) {
                BackHandler(enabled = selectedBook != null) {
                    selectedBook = null
                }

                BackHandler(enabled = selectedBook == null && route != "home") {
                    route = "home"
                }

                Box(modifier = Modifier.fillMaxSize()) {
                    if (selectedBook != null) {
                        ReaderScreen(
                            book = selectedBook!!,
                            currentAccent = accentTheme,
                            onAccentChange = viewModel::setAccentTheme,
                            onBack = { selectedBook = null },
                            onUpdateProgress = viewModel::updateBookProgress
                        )
                    } else {
                        Box(modifier = Modifier.fillMaxSize()) {
                            // Keep the backdrop and screen content in one source-ready subtree.
                            // A future backdrop effect can capture this container once, while the
                            // navigation bar remains a separate overlay drawn above it.
                            Box(modifier = Modifier.fillMaxSize()) {
                                AmbientBackdrop(modifier = Modifier.fillMaxSize())
                                AnimatedContent(
                                    targetState = route,
                                    transitionSpec = {
                                        if (!motionEnabled) {
                                            EnterTransition.None togetherWith ExitTransition.None
                                        } else {
                                            val direction =
                                                if (routeOrder.indexOf(targetState) > routeOrder.indexOf(initialState)) 1 else -1
                                            (slideInHorizontally(animationSpec = tween(260)) { width -> direction * width / 7 } +
                                                fadeIn(animationSpec = tween(220)))
                                                .togetherWith(
                                                    slideOutHorizontally(animationSpec = tween(220)) { width -> -direction * width / 7 } +
                                                        fadeOut(animationSpec = tween(120))
                                                )
                                        }
                                    },
                                    modifier = Modifier.fillMaxSize(),
                                    label = "screenTransition"
                                ) { target ->
                                    when (target) {
                                        "home" -> HomeScreen(
                                            books = books,
                                            isLoading = booksState == null,
                                            onToggleFavorite = viewModel::toggleFavorite,
                                            onBookClick = { selectedBook = it },
                                            onImportClick = { importLauncher.launch(importMimeTypes) }
                                        )
                                        "explore" -> ExploreScreen(
                                            onImportClick = { importLauncher.launch(importMimeTypes) }
                                        )
                                        "bookmarks" -> BookmarksScreen(
                                            favorites = favorites,
                                            isLoading = booksState == null,
                                            onToggleFavorite = viewModel::toggleFavorite,
                                            onBookClick = { selectedBook = it },
                                            onImportClick = { importFavoriteLauncher.launch(importMimeTypes) }
                                        )
                                        "settings" -> SettingsScreen(
                                            themeMode = themeMode,
                                            onThemeModeChange = viewModel::setThemeMode,
                                            currentAccent = accentTheme,
                                            onAccentChange = viewModel::setAccentTheme
                                        )
                                    }
                                }
                            }

                            FloatingActionNavBar(
                                selectedRoute = route,
                                onRouteSelected = { route = it },
                                onAddClick = {
                                    if (route == "bookmarks") importFavoriteLauncher.launch(importMimeTypes)
                                    else importLauncher.launch(importMimeTypes)
                                },
                                modifier = Modifier.align(Alignment.BottomCenter)
                            )
                        }
                    }
                }
            }
        }
    }
}

package com.example.oryn.reader

import android.app.Application
import android.content.Intent
import android.net.Uri
import android.widget.Toast
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.example.oryn.reader.data.importer.BookMetadataReader
import com.example.oryn.reader.data.local.OrynDatabase
import com.example.oryn.reader.data.local.entity.BookEntity
import com.example.oryn.reader.data.local.preferences.UserPreferencesRepository
import com.example.oryn.reader.data.repository.BookRepository
import com.example.oryn.reader.ui.theme.AppAccentTheme
import com.example.oryn.reader.ui.theme.AppThemeMode
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class MainViewModel(application: Application) : AndroidViewModel(application) {

    private val repository = BookRepository(
        OrynDatabase.getDatabase(application).bookDao()
    )
    private val metadataReader = BookMetadataReader(application)
    private val prefsRepository = UserPreferencesRepository(application)

    val books: StateFlow<List<BookEntity>?> = repository.allBooks
        .map { it as List<BookEntity>? }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), null)

    val favoriteBooks: StateFlow<List<BookEntity>> = repository.favoriteBooks
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), emptyList())

    val themeMode: StateFlow<AppThemeMode> = prefsRepository.themeModeFlow
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), AppThemeMode.SYSTEM)

    val dyslexicFont: StateFlow<Boolean> = prefsRepository.dyslexicFontFlow
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), false)

    val accentTheme: StateFlow<AppAccentTheme> = prefsRepository.accentThemeFlow
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), AppAccentTheme.ORANGE)

    init {
        viewModelScope.launch { repository.deleteSamples() }
    }

    fun setDarkMode(enabled: Boolean) {
        viewModelScope.launch { prefsRepository.setDarkMode(enabled) }
    }

    fun setThemeMode(mode: AppThemeMode) {
        viewModelScope.launch { prefsRepository.setThemeMode(mode) }
    }

    fun setDyslexicFont(enabled: Boolean) {
        viewModelScope.launch { prefsRepository.setDyslexicFont(enabled) }
    }

    fun setAccentTheme(theme: AppAccentTheme) {
        viewModelScope.launch { prefsRepository.setAccentTheme(theme) }
    }

    fun importBooks(uris: List<Uri>, favorite: Boolean = false) {
        if (uris.isEmpty()) return
        val app = getApplication<Application>()

        viewModelScope.launch(Dispatchers.IO) {
            var added = 0
            var duplicates = 0
            var unsupported = 0
            var failed = 0

            for (uri in uris) {
                try {
                    val path = uri.toString()
                    if (repository.exists(path)) {
                        duplicates++
                        continue
                    }

                    val meta = metadataReader.read(uri)
                    if (meta == null) {
                        unsupported++
                        continue
                    }

                    app.contentResolver.takePersistableUriPermission(
                        uri, Intent.FLAG_GRANT_READ_URI_PERMISSION
                    )
                    repository.addBook(
                        BookEntity(
                            title = meta.title,
                            author = meta.author,
                            filePath = path,
                            format = meta.format,
                            isFavorite = favorite
                        )
                    )
                    added++
                } catch (_: SecurityException) {
                    failed++
                } catch (_: Exception) {
                    failed++
                }
            }

            val summary = buildList {
                if (added > 0) add("$added importé(s)")
                if (duplicates > 0) add("$duplicates déjà présent(s)")
                if (unsupported > 0) add("$unsupported format(s) non pris en charge")
                if (failed > 0) add("$failed échec(s) d’accès ou d’import")
            }.joinToString(" · ").ifBlank { "Aucun livre importé." }
            val message = "Importation : $summary"
            withContext(Dispatchers.Main) {
                Toast.makeText(app, message, Toast.LENGTH_SHORT).show()
            }
        }
    }

    fun updateBookProgress(book: BookEntity, progress: Float) {
        viewModelScope.launch {
            repository.updateBook(
                book.copy(
                    progress = progress.coerceIn(0f, 1f),
                    lastReadTimestamp = System.currentTimeMillis()
                )
            )
        }
    }

    fun toggleFavorite(book: BookEntity) {
        viewModelScope.launch { repository.setFavorite(book, !book.isFavorite) }
    }

    fun deleteBook(book: BookEntity) {
        viewModelScope.launch { repository.deleteBook(book) }
    }
}

@Composable
fun OrynTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    accentTheme: AppAccentTheme = AppAccentTheme.ORANGE,
    content: @Composable () -> Unit
) {
    val context = LocalContext.current
    val useDynamic = accentTheme == AppAccentTheme.DYNAMIC && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S

    val colorScheme = when {
        useDynamic -> {
            val dynamic = if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
            val primary = accessibleAccentColor(dynamic.primary, darkTheme)
            dynamic.copy(
                primary = primary,
                onPrimary = contrastingContent(primary),
                onPrimaryContainer = contrastingContent(dynamic.primaryContainer),
                onSecondary = contrastingContent(dynamic.secondary),
                onTertiary = contrastingContent(dynamic.tertiary)
            )
        }
        else -> {
            val effectiveAccent = if (accentTheme == AppAccentTheme.DYNAMIC) AppAccentTheme.SALMON else accentTheme
            val sourcePrimary = if (darkTheme) effectiveAccent.primaryDark else effectiveAccent.primaryLight
            val primary = accessibleAccentColor(sourcePrimary, darkTheme)
            val secondary = lerp(primary, if (darkTheme) Color.White else Color.Black, 0.12f)
            val tertiary = lerp(primary, if (darkTheme) Color(0xFFFFD59E) else Color(0xFF66558F), 0.22f)
            val primaryContainer = lerp(primary, if (darkTheme) Color(0xFF171717) else Color.White, if (darkTheme) 0.52f else 0.84f)
            if (darkTheme) {
                darkColorScheme(
                    primary = primary,
                    onPrimary = contrastingContent(primary),
                    primaryContainer = primaryContainer,
                    onPrimaryContainer = contrastingContent(primaryContainer),
                    secondary = secondary,
                    onSecondary = contrastingContent(secondary),
                    tertiary = tertiary,
                    onTertiary = contrastingContent(tertiary)
                )
            } else {
                lightColorScheme(
                    primary = primary,
                    onPrimary = contrastingContent(primary),
                    primaryContainer = primaryContainer,
                    onPrimaryContainer = contrastingContent(primaryContainer),
                    secondary = secondary,
                    onSecondary = contrastingContent(secondary),
                    tertiary = tertiary,
                    onTertiary = contrastingContent(tertiary)
                )
            }
        }
    }

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            WindowCompat.getInsetsController(window, view).apply {
                isAppearanceLightStatusBars = !darkTheme
                isAppearanceLightNavigationBars = !darkTheme
            }
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        content = content
    )
}

/** Keeps the primary accent readable when used as foreground content on app surfaces. */
fun accessibleAccentColor(accent: Color, darkTheme: Boolean): Color {
    val surface = if (darkTheme) Color(0xFF141218) else Color.White
    val destination = if (darkTheme) Color.White else Color.Black
    fun contrast(foreground: Color): Float {
        val first = foreground.luminance()
        val second = surface.luminance()
        return (maxOf(first, second) + 0.05f) / (minOf(first, second) + 0.05f)
    }
    if (contrast(accent) >= 4.5f) return accent

    var low = 0f
    var high = 1f
    repeat(16) {
        val fraction = (low + high) / 2f
        if (contrast(lerp(accent, destination, fraction)) >= 4.5f) high = fraction else low = fraction
    }
    return lerp(accent, destination, high)
}

private fun contrastingContent(background: Color): Color {
    val black = Color.Black
    val white = Color.White
    fun contrastRatio(foreground: Color): Float {
        val first = foreground.luminance()
        val second = background.luminance()
        return (maxOf(first, second) + 0.05f) / (minOf(first, second) + 0.05f)
    }
    return if (contrastRatio(black) >= contrastRatio(white)) black else white
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun ReaderScreen(
    book: BookEntity,
    currentAccent: AppAccentTheme,
    onAccentChange: (AppAccentTheme) -> Unit,
    onBack: () -> Unit,
    onUpdateProgress: (BookEntity, Float) -> Unit
) {
    val context = LocalContext.current
    val activity = context as? Activity
    val view = LocalView.current
    val scope = rememberCoroutineScope()
    val preferencesRepo = remember { ReaderPreferencesRepository(context) }
    val settings by preferencesRepo.readerSettingsFlow.collectAsState(initial = ReaderSettings())

    var loadResult by remember { mutableStateOf<BookLoadResult?>(null) }
    var uiVisible by remember { mutableStateOf(false) }
    var showSheet by remember { mutableStateOf(false) }

    LaunchedEffect(book) {
        loadResult = BookTextLoader(context).loadBook(book.filePath, book.format)
    }

    LaunchedEffect(settings.brightness, settings.autoBrightness) {
        activity?.window?.attributes = activity?.window?.attributes?.apply {
            screenBrightness = if (settings.autoBrightness) {
                android.view.WindowManager.LayoutParams.BRIGHTNESS_OVERRIDE_NONE
            } else {
                settings.brightness.coerceIn(0.01f, 1.0f)
            }
        }
    }

    LaunchedEffect(settings.hideStatusBar, uiVisible, showSheet) {
        val window = activity?.window ?: return@LaunchedEffect
        val controller = WindowInsetsControllerCompat(window, view)
        if (settings.hideStatusBar && !uiVisible && !showSheet) {
            controller.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
            controller.hide(WindowInsetsCompat.Type.statusBars())
        } else {
            controller.show(WindowInsetsCompat.Type.statusBars())
        }
    }

    DisposableEffect(activity, view) {
        onDispose {
            activity?.window?.let { window ->
                WindowInsetsControllerCompat(window, view).show(WindowInsetsCompat.Type.statusBars())
            }
        }
    }

    val backgroundColor = when (settings.themePreset) {
        ReaderThemePreset.LIGHT -> Color(0xFFFBF9F5)
        ReaderThemePreset.BEIGE -> Color(0xFFF4ECD8)
        ReaderThemePreset.MIDNIGHT -> Color(0xFF1E1E1E)
        ReaderThemePreset.OLED -> Color.Black
    }

    val foregroundColor = when (settings.themePreset) {
        ReaderThemePreset.LIGHT -> Color(0xFF1E1E1E)
        ReaderThemePreset.BEIGE -> Color(0xFF3C3024)
        ReaderThemePreset.MIDNIGHT -> Color(0xFFE0E0E0)
        ReaderThemePreset.OLED -> Color(0xFFF0F0F0)
    }

    val fontFamily = when (settings.fontFamily) {
        ReaderFontFamily.SYSTEM -> FontFamily.Default
        ReaderFontFamily.SERIF -> FontFamily.Serif
        ReaderFontFamily.SANS_SERIF -> FontFamily.SansSerif
        ReaderFontFamily.DYSLEXIC -> ReaderOpenDyslexicFont ?: FontFamily.Default
    }

    val horizontalPadding = when (settings.marginSize) {
        ReaderMarginSize.NARROW -> 16.dp
        ReaderMarginSize.NORMAL -> 36.dp
        ReaderMarginSize.WIDE -> 56.dp
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(backgroundColor)
    ) {
        when (val result = loadResult) {
            null -> {
                Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    CircularProgressIndicator(color = foregroundColor)
                }
            }
            is BookLoadResult.Error -> {
                Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    Text(text = result.message, color = foregroundColor, style = MaterialTheme.typography.bodyLarge)
                }
            }
            is BookLoadResult.Success -> {
                val isPdf = result.bitmaps.isNotEmpty()
                val totalPages = if (isPdf) result.bitmaps.size else result.pages.size
                val pagerState = rememberPagerState(pageCount = { totalPages })

                LaunchedEffect(pagerState.currentPage) {
                    if (totalPages > 0) {
                        val progress = (pagerState.currentPage + 1).toFloat() / totalPages.toFloat()
                        onUpdateProgress(book, progress)
                    }
                }

                HorizontalPager(
                    state = pagerState,
                    modifier = Modifier.fillMaxSize()
                ) { page ->
                    val pageOffset = (pagerState.currentPage - page) + pagerState.currentPageOffsetFraction

                    Box(
                        modifier = Modifier
                            .fillMaxSize()
                            .graphicsLayer {
                                if (settings.scrollMode == ReaderScrollMode.PAGES) {
                                    cameraDistance = 15f * density
                                    rotationY = pageOffset * -75f
                                    alpha = (1f - pageOffset.absoluteValue * 0.4f).coerceIn(0.2f, 1f)
                                }
                            }
                            .clickable(
                                interactionSource = remember { MutableInteractionSource() },
                                indication = null
                            ) {
                                uiVisible = !uiVisible
                                if (uiVisible) showSheet = false
                            }
                            .padding(horizontal = horizontalPadding, vertical = 70.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Column(
                            modifier = Modifier.fillMaxSize(),
                            verticalArrangement = Arrangement.SpaceBetween,
                            horizontalAlignment = Alignment.CenterHorizontally
                        ) {
                            if (isPdf) {
                                Image(
                                    bitmap = result.bitmaps[page].asImageBitmap(),
                                    contentDescription = "Page ${page + 1}",
                                    modifier = Modifier
                                        .weight(1f)
                                        .fillMaxWidth()
                                        .clip(RoundedCornerShape(8.dp)),
                                    contentScale = ContentScale.Fit
                                )
                            } else {
                                Text(
                                    text = result.pages[page],
                                    style = MaterialTheme.typography.bodyLarge.copy(
                                        fontFamily = fontFamily,
                                        fontSize = settings.fontSizePt.sp,
                                        color = foregroundColor,
                                        lineHeight = (settings.fontSizePt * settings.lineSpacing).sp,
                                        textAlign = if (settings.alignLeft) TextAlign.Start else TextAlign.Justify
                                    ),
                                    modifier = Modifier.weight(1f)
                                )
                            }

                            Row(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(top = 12.dp),
                                horizontalArrangement = Arrangement.Center
                            ) {
                                Text(
                                    text = "${page + 1} / $totalPages",
                                    style = MaterialTheme.typography.bodySmall,
                                    color = foregroundColor.copy(alpha = 0.5f),
                                    fontFamily = fontFamily
                                )
                            }
                        }
                    }
                }
            }
        }

        // TopBar épurée
        AnimatedVisibility(
            visible = uiVisible,
            enter = fadeIn(),
            exit = fadeOut(),
            modifier = Modifier.align(Alignment.TopCenter)
        ) {
            Surface(
                color = backgroundColor.copy(alpha = 0.95f),
                tonalElevation = 0.dp
            ) {
                TopAppBar(
                    title = {
                        Column {
                            Text(book.title, color = foregroundColor, style = MaterialTheme.typography.titleMedium, maxLines = 1)
                            Text(book.author, color = foregroundColor.copy(alpha = 0.6f), style = MaterialTheme.typography.bodySmall, maxLines = 1)
                        }
                    },
                    navigationIcon = {
                        IconButton(onClick = onBack) {
                            Icon(Icons.Rounded.ArrowBack, contentDescription = "Retour", tint = foregroundColor)
                        }
                    },
                    actions = {
                        IconButton(onClick = { showSheet = !showSheet }) {
                            Icon(Icons.Rounded.Settings, contentDescription = "Réglages", tint = foregroundColor)
                        }
                    },
                    colors = TopAppBarDefaults.topAppBarColors(containerColor = Color.Transparent)
                )
            }
        }

        // Panneau Flottant superposé en bas
        AnimatedVisibility(
            visible = uiVisible && showSheet,
            enter = fadeIn() + slideInVertically { it },
            exit = fadeOut() + slideOutVertically { it },
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .padding(bottom = 16.dp)
        ) {
            ReaderControlsSheet(
                settings = settings,
                onSettingsChange = { updater ->
                    scope.launch {
                        preferencesRepo.updateSettings(updater)
                    }
                },
                currentChapterTitle = book.title,
                currentAccent = currentAccent,
                onAccentChange = onAccentChange
            )
        }
    }
}

internal fun systemAnimationsEnabled(context: Context): Boolean =
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {

    } else {
        runCatching {
            Settings.Global.getFloat(

                1f
            ) > 0f
        }.getOrDefault(true)
    }

package com.example.oryn.reader.ui.components

import androidx.compose.ui.geometry.Size
import androidx.compose.ui.graphics.Outline
import androidx.compose.ui.graphics.Path
import androidx.compose.ui.graphics.Shape
import androidx.compose.ui.unit.Density
import androidx.compose.ui.unit.LayoutDirection
import kotlin.math.abs
import kotlin.math.cos
import kotlin.math.sin
import kotlin.math.sqrt
import kotlin.math.sign

/** Continuous superellipse (n=4), used for Oryn's expressive squircle surfaces. */
object OrynSquircleShape : Shape {
    override fun createOutline(size: Size, layoutDirection: LayoutDirection, density: Density): Outline {
        val halfWidth = size.width / 2f
        val halfHeight = size.height / 2f
        if (halfWidth <= 0f || halfHeight <= 0f) return Outline.Rectangle(androidx.compose.ui.geometry.Rect(0f, 0f, size.width, size.height))

        val path = Path()
        val segments = 96
        for (index in 0..segments) {
            val angle = (2.0 * Math.PI * index / segments)
            val x = halfWidth + halfWidth * sign(cos(angle)).toFloat() * sqrt(abs(cos(angle))).toFloat()
            val y = halfHeight + halfHeight * sign(sin(angle)).toFloat() * sqrt(abs(sin(angle))).toFloat()
            if (index == 0) path.moveTo(x, y) else path.lineTo(x, y)
        }
        path.close()
        return Outline.Generic(path)
    }
}

@Composable
fun AccentPaletteOption(
    accent: AppAccentTheme,
    selected: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    val darkTheme = MaterialTheme.colorScheme.background.luminance() < 0.5f
    val paletteColor = when (accent) {
        AppAccentTheme.DYNAMIC -> MaterialTheme.colorScheme.primary
        else -> if (darkTheme) accent.primaryDark else accent.primaryLight
    }
    val base = accessibleAccentColor(
        paletteColor.takeUnless { it == Color.Unspecified } ?: MaterialTheme.colorScheme.primary,
        darkTheme
    )
    val animationsEnabled = systemAnimationsEnabled(LocalContext.current)
    val scale by animateFloatAsState(
        targetValue = if (selected) 1.06f else 1f,
        animationSpec = if (animationsEnabled) spring(dampingRatio = 0.76f, stiffness = 520f) else snap(),
        label = "accentPaletteScale"
    )
    val petalColors = listOf(
        lerp(base, Color.White, 0.20f), base,
        lerp(base, Color.Black, 0.13f), lerp(base, Color.White, 0.38f)
    )
    val description = accent.label

    Column(
        modifier = modifier
            .width(82.dp)
            .clip(OrynSquircleShape)
            .selectable(selected = selected, role = Role.RadioButton, onClick = onClick)
            .semantics { contentDescription = description }
            .padding(vertical = 4.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Box(
            modifier = Modifier.size(52.dp).graphicsLayer { scaleX = scale; scaleY = scale },
            contentAlignment = Alignment.Center
        ) {
            val petal = OrynSquircleShape
            Surface(Modifier.offset(y = (-9).dp).size(20.dp, 27.dp), petal, petalColors[0]) {}
            Surface(Modifier.offset(x = 9.dp).size(20.dp, 27.dp).graphicsLayer { rotationZ = 90f }, petal, petalColors[1]) {}
            Surface(Modifier.offset(y = 9.dp).size(20.dp, 27.dp).graphicsLayer { rotationZ = 180f }, petal, petalColors[2]) {}
            Surface(Modifier.offset(x = (-9).dp).size(20.dp, 27.dp).graphicsLayer { rotationZ = 270f }, petal, petalColors[3]) {}
            if (selected) {
                Surface(Modifier.size(23.dp), CircleShape, color = MaterialTheme.colorScheme.surface) {
                    Box(contentAlignment = Alignment.Center) {
                        Icon(
                            Icons.Rounded.Check,
                            contentDescription = null,
                            tint = bestContrastingInk(MaterialTheme.colorScheme.surface),
                            modifier = Modifier.size(17.dp)
                        )
                    }
                }
            }
        }
        Text(
            text = accent.label,
            style = MaterialTheme.typography.labelMedium,
            color = if (selected) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
            fontWeight = if (selected) FontWeight.SemiBold else FontWeight.Normal,
            textAlign = TextAlign.Center,
        )
    }
}

private fun bestContrastingInk(background: Color): Color {
    fun ratio(foreground: Color): Float {
        val first = foreground.luminance()
        val second = background.luminance()
        return (maxOf(first, second) + 0.05f) / (minOf(first, second) + 0.05f)
    }
    return if (ratio(Color.Black) >= ratio(Color.White)) Color.Black else Color.White
}

@Composable
fun FloatingActionNavBar(
    selectedRoute: String,
    onRouteSelected: (String) -> Unit,
    onFilesPicked: (List<Uri>) -> Unit,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val context = LocalContext.current

    val picker = rememberLauncherForActivityResult(
        ActivityResultContracts.OpenMultipleDocuments()
    ) { uris ->
        if (uris.isNotEmpty()) {
            uris.forEach { uri ->
                try {
                    context.contentResolver.takePersistableUriPermission(
                        uri, Intent.FLAG_GRANT_READ_URI_PERMISSION
                    )
                } catch (_: SecurityException) {}
            }
            onFilesPicked(uris)
        }
    }

    val barHeight = 72.dp
    val fabOverflow = 46.dp
    val barShape = remember {
        NotchedBarShape(corner = 28.dp, notchHalfWidth = 52.dp, notchDepth = 24.dp)
    }

    Box(
        modifier
            .fillMaxWidth()
            .navigationBarsPadding()
            .padding(horizontal = 16.dp, vertical = 12.dp)
            .height(barHeight + fabOverflow)
    ) {
        Surface(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth()
                .height(barHeight),
            shape = barShape,
            color = colors.surfaceContainerHigh,
            contentColor = colors.onSurfaceVariant,
            shadowElevation = 10.dp,
            border = BorderStroke(1.dp, colors.outlineVariant.copy(alpha = 0.5f)),
        ) {
            Row(
                Modifier
                    .fillMaxSize()
                    .padding(horizontal = 12.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                val tabs = ReaderTab.entries
                tabs.take(2).forEach { tab ->
                    val isSelected = selectedRoute == tab.route
                    NavTab(tab, isSelected, { onRouteSelected(tab.route) }, Modifier.weight(1f))
                }
                Spacer(Modifier.width(72.dp))
                tabs.drop(2).forEach { tab ->
                    val isSelected = selectedRoute == tab.route
                    NavTab(tab, isSelected, { onRouteSelected(tab.route) }, Modifier.weight(1f))
                }
            }
        }

        CenterAddButton(
            onClick = { picker.launch(READER_MIME_TYPES) },
            modifier = Modifier.align(Alignment.TopCenter),
        )
    }
}

private class NotchedBarShape(
    private val corner: Dp,
    private val notchHalfWidth: Dp,
    private val notchDepth: Dp,
) : Shape {
    override fun createOutline(
        size: Size,
        layoutDirection: LayoutDirection,
        density: Density,
    ): Outline {
        val r = with(density) { corner.toPx() }
        val hw = with(density) { notchHalfWidth.toPx() }
        val d = with(density) { notchDepth.toPx() }
        val w = size.width
        val h = size.height
        val cx = w / 2f

        val path = Path().apply {
            moveTo(r, 0f)
            lineTo(cx - hw, 0f)
            cubicTo(cx - hw * 0.6f, 0f, cx - hw * 0.4f, d, cx, d)
            cubicTo(cx + hw * 0.4f, d, cx + hw * 0.6f, 0f, cx + hw, 0f)
            lineTo(w - r, 0f)
            arcTo(Rect(w - 2 * r, 0f, w, 2 * r), -90f, 90f, false)
            lineTo(w, h - r)
            arcTo(Rect(w - 2 * r, h - 2 * r, w, h), 0f, 90f, false)
            lineTo(r, h)
            arcTo(Rect(0f, h - 2 * r, 2 * r, h), 90f, 90f, false)
            lineTo(0f, r)
            arcTo(Rect(0f, 0f, 2 * r, 2 * r), 180f, 90f, false)
            close()
        }
        return Outline.Generic(path)
    }
}

private val Context.dataStore by preferencesDataStore(name = "user_settings")

class UserPreferencesRepository(private val context: Context) {

    private object PreferencesKeys {
        val DARK_MODE = booleanPreferencesKey("dark_mode")
        val THEME_MODE = stringPreferencesKey("theme_mode")
        val DYSLEXIC_FONT = booleanPreferencesKey("dyslexic_font")
        val ACCENT_THEME = stringPreferencesKey("accent_theme")
    }

    val themeModeFlow: Flow<AppThemeMode> = context.dataStore.data.map { prefs ->
        prefs[PreferencesKeys.THEME_MODE]?.let { runCatching { AppThemeMode.valueOf(it) }.getOrNull() }
            ?: if (prefs[PreferencesKeys.DARK_MODE] == true) AppThemeMode.DARK else AppThemeMode.LIGHT
    }

    val dyslexicFontFlow: Flow<Boolean> = context.dataStore.data.map { prefs ->
        prefs[PreferencesKeys.DYSLEXIC_FONT] ?: false
    }

    val accentThemeFlow: Flow<AppAccentTheme> = context.dataStore.data.map { prefs ->
        val name = prefs[PreferencesKeys.ACCENT_THEME] ?: AppAccentTheme.ORANGE.name
        runCatching { AppAccentTheme.valueOf(name) }.getOrDefault(AppAccentTheme.ORANGE)
    }

    suspend fun setDarkMode(enabled: Boolean) {
        context.dataStore.edit { prefs ->
            prefs[PreferencesKeys.DARK_MODE] = enabled
            prefs[PreferencesKeys.THEME_MODE] = if (enabled) AppThemeMode.DARK.name else AppThemeMode.LIGHT.name
        }
    }

    suspend fun setThemeMode(mode: AppThemeMode) {
        context.dataStore.edit { prefs ->
            prefs[PreferencesKeys.THEME_MODE] = mode.name
            if (mode != AppThemeMode.SYSTEM) prefs[PreferencesKeys.DARK_MODE] = mode == AppThemeMode.DARK
        }
    }

    suspend fun setDyslexicFont(enabled: Boolean) {
        context.dataStore.edit { prefs ->
            prefs[PreferencesKeys.DYSLEXIC_FONT] = enabled
        }
    }

    suspend fun setAccentTheme(theme: AppAccentTheme) {
        context.dataStore.edit { prefs ->
            prefs[PreferencesKeys.ACCENT_THEME] = theme.name
        }
    }
}

package com.example.oryn.reader.data.importer

import android.content.Context
import android.net.Uri
import android.provider.OpenableColumns
import android.util.Xml
import org.xmlpull.v1.XmlPullParser
import java.io.InputStream
import java.util.zip.ZipInputStream

data class BookMetadata(
    val title: String,
    val author: String,
    val format: String
)

/** Lit titre / auteur / format d'un fichier choisi via le sélecteur Android. */
class BookMetadataReader(private val context: Context) {

    /** Renvoie null si le format n'est pas pris en charge. */
    fun read(uri: Uri): BookMetadata? {
        val fileName = displayName(uri) ?: "Livre"
        val extension = fileName.substringAfterLast('.', "").lowercase()
        val mime = context.contentResolver.getType(uri)

        val format = when {
            extension == "epub" || mime == "application/epub+zip" -> "EPUB"
            extension == "pdf" || mime == "application/pdf" -> "PDF"
            extension == "txt" || mime == "text/plain" -> "TXT"
            else -> return null
        }

        val fallbackTitle = fileName.substringBeforeLast('.')
            .replace('_', ' ')
            .replace('-', ' ')
            .trim()
            .ifBlank { "Livre sans titre" }

        val epub = if (format == "EPUB") readEpub(uri) else null

        return BookMetadata(
            title = epub?.first ?: fallbackTitle,
            author = epub?.second ?: "Auteur inconnu",
            format = format
        )
    }

    private fun displayName(uri: Uri): String? =
        runCatching {
            context.contentResolver
                .query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)
                ?.use { cursor -> if (cursor.moveToFirst()) cursor.getString(0) else null }
        }.getOrNull()

    // ── EPUB : container.xml -> fichier OPF -> dc:title / dc:creator ──
    private fun readEpub(uri: Uri): Pair<String?, String?>? = runCatching {
        val opfPath = openZip(uri) { zip ->
            if (findEntry(zip, "META-INF/container.xml")) rootfilePath(zip) else null
        } ?: return@runCatching null

        openZip(uri) { zip ->
            if (findEntry(zip, opfPath)) parseOpf(zip) else null
        }
    }.getOrNull()

    private fun <T> openZip(uri: Uri, block: (ZipInputStream) -> T?): T? =
        context.contentResolver.openInputStream(uri)?.use { input ->
            ZipInputStream(input).use(block)
        }

    private fun findEntry(zip: ZipInputStream, name: String): Boolean {
        while (true) {
            val entry = zip.nextEntry ?: return false
            if (entry.name == name) return true
        }
    }

    private fun rootfilePath(stream: InputStream): String? {
        val parser = Xml.newPullParser()
        parser.setInput(stream, null)
        while (parser.eventType != XmlPullParser.END_DOCUMENT) {
            if (parser.eventType == XmlPullParser.START_TAG &&
                parser.name.substringAfter(':') == "rootfile"
            ) {
                return parser.getAttributeValue(null, "full-path")
            }
            parser.next()
        }
        return null
    }

    private fun parseOpf(stream: InputStream): Pair<String?, String?> {
        val parser = Xml.newPullParser()
        parser.setInput(stream, null)
        var title: String? = null
        var author: String? = null
        while (parser.eventType != XmlPullParser.END_DOCUMENT && (title == null || author == null)) {
            if (parser.eventType == XmlPullParser.START_TAG) {
                when (parser.name.substringAfter(':')) {
                    "title" -> if (title == null) {
                        title = parser.nextText().trim().takeIf { it.isNotBlank() }
                    }
                    "creator" -> if (author == null) {
                        author = parser.nextText().trim().takeIf { it.isNotBlank() }
                    }
                }
            }
            parser.next()
        }
        return title to author
    }
}

package com.example.oryn.reader.data.importer

import android.content.Context
import android.graphics.Bitmap
import android.graphics.pdf.PdfRenderer
import android.net.Uri
import android.os.ParcelFileDescriptor
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.BufferedReader
import java.io.InputStreamReader
import java.util.zip.ZipInputStream

sealed interface BookLoadResult {
    data class Success(val pages: List<String>, val bitmaps: List<Bitmap> = emptyList()) : BookLoadResult
    data class Error(val message: String) : BookLoadResult
}

class BookTextLoader(private val context: Context) {

    suspend fun loadBook(filePath: String, format: String): BookLoadResult = withContext(Dispatchers.IO) {
        if (filePath.isBlank()) return@withContext BookLoadResult.Error("Chemin de fichier invalide.")
        val uri = Uri.parse(filePath)

        runCatching {
            when (format.uppercase()) {
                "TXT" -> {
                    val text = loadTxt(uri)?.takeIf { it.isNotBlank() }
                        ?: return@runCatching BookLoadResult.Error("Le fichier texte est vide ou inaccessible.")
                    val pages = chunkText(text, 1500)
                    BookLoadResult.Success(pages)
                }
                "EPUB" -> {
                    val chapters = loadEpubChapters(uri)
                    BookLoadResult.Success(chapters)
                }
                "PDF" -> {
                    loadPdfPages(uri)
                }
                else -> BookLoadResult.Error("Format non supporté pour la lecture directe.")
            }
        }.getOrElse { BookLoadResult.Error("Erreur de chargement : ${it.localizedMessage}") }
    }

    private fun loadTxt(uri: Uri): String? {
        context.contentResolver.openInputStream(uri)?.use { inputStream ->
            BufferedReader(InputStreamReader(inputStream, Charsets.UTF_8)).use { reader ->
                return reader.readText()
            }
        }
        return null
    }

    private fun loadEpubChapters(uri: Uri): List<String> {
        val chapters = mutableListOf<String>()
        context.contentResolver.openInputStream(uri)?.use { inputStream ->
            ZipInputStream(inputStream).use { zip ->
                var entry = zip.nextEntry
                while (entry != null) {
                    val name = entry.name.lowercase()
                    if ((name.endsWith(".html") || name.endsWith(".xhtml") || name.endsWith(".htm")) &&
                        !name.contains("toc") && !name.contains("cover") && !name.contains("title")) {
                        
                        val content = zip.bufferedReader(Charsets.UTF_8).readText()
                        val cleanText = content
                            .replace(Regex("(?is)<style.*?>.*?</style>"), "")
                            .replace(Regex("(?is)<script.*?>.*?</script>"), "")
                            .replace(Regex("(?is)<head.*?>.*?</head>"), "")
                            .replace(Regex("<[^>]*>"), " ")
                            .replace(Regex("&nbsp;"), " ")
                            .replace(Regex("&amp;"), "&")
                            .replace(Regex("&quot;"), "\"")
                            .replace(Regex("[ \\t]+"), " ")
                            .replace(Regex("\n\\s*\n"), "\n\n")
                            .trim()

                        if (cleanText.length > 30) {
                            chapters.add(cleanText)
                        }
                    }
                    zip.closeEntry()
                    entry = zip.nextEntry
                }
            }
        }
        return if (chapters.isEmpty()) listOf("Aucun contenu textuel lisible trouvé dans cet EPUB.") else chapters
    }

    private fun loadPdfPages(uri: Uri): BookLoadResult {
        val descriptor = context.contentResolver.openFileDescriptor(uri, "r")
            ?: return BookLoadResult.Error("Impossible d'ouvrir le document PDF.")

        descriptor.use { fd ->
            PdfRenderer(fd).use { renderer ->
                val pages = mutableListOf<String>()
                val bitmaps = mutableListOf<Bitmap>()

                for (index in 0 until renderer.pageCount) {
                    renderer.openPage(index).use { page ->
                        val width = minOf(page.width, 1200)
                        val height = (page.height.toFloat() * width / page.width)
                            .toInt().coerceAtLeast(1)
                        val bitmap = Bitmap.createBitmap(
                            width, height, Bitmap.Config.ARGB_8888
                        )
                        bitmap.eraseColor(android.graphics.Color.WHITE)
                        page.render(
                            bitmap, null, null,
                            PdfRenderer.Page.RENDER_MODE_FOR_DISPLAY
                        )
                        bitmaps.add(bitmap)
                        pages.add("Page ${index + 1} sur ${renderer.pageCount}")
                    }
                }

                if (pages.isEmpty()) {
                    return BookLoadResult.Error("Ce PDF ne contient aucune page.")
                }
                return BookLoadResult.Success(pages, bitmaps)
            }
        }
    }

    private fun chunkText(text: String, maxCharsPerPage: Int): List<String> {
        if (text.length <= maxCharsPerPage) return listOf(text)
        val chunks = mutableListOf<String>()
        var start = 0
        while (start < text.length) {
            val end = minOf(start + maxCharsPerPage, text.length)
            chunks.add(text.substring(start, end).trim())
            start = end
        }
        return chunks
    }
}

    var loadResult by remember { mutableStateOf<BookLoadResult>(BookLoadResult.Success(listOf("Chargement en cours..."))) }
    
    var uiVisible by remember { mutableStateOf(false) }
    var textSizeValue by remember { mutableStateOf(18f) }
    val textSize = textSizeValue.sp
    var currentTheme by remember { mutableStateOf(ReaderTheme.LIGHT) }
    var showSettingsMenu by remember { mutableStateOf(false) }

    LaunchedEffect(book) {
        loadResult = BookTextLoader(context).loadBook(book.filePath, book.format)
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(currentTheme.bg)
    ) {

            is BookLoadResult.Error -> {
                Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    Text(text = result.message, color = currentTheme.fg, style = MaterialTheme.typography.bodyLarge)
                }

            is BookLoadResult.Success -> {
                val pages = result.pages
                val pagerState = rememberPagerState(pageCount = { pages.size })

                // Sauvegarder la progression de lecture en temps réel
                LaunchedEffect(pagerState.currentPage) {
                    if (pages.isNotEmpty()) {
                        val progress = (pagerState.currentPage + 1).toFloat() / pages.size.toFloat()
                        onUpdateProgress(book, progress)
                    }

                    state = pagerState,

                                // Effet 3D Page Turn / Curl (Rotation Y et profondeur)

package com.example.oryn.reader.data.importer

import android.content.Context
import android.graphics.Bitmap
import android.graphics.pdf.PdfRenderer
import android.net.Uri
import android.os.ParcelFileDescriptor
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.BufferedReader
import java.io.InputStreamReader
import java.util.zip.ZipInputStream

sealed interface BookLoadResult {
    data class Success(val pages: List<String>, val bitmaps: List<Bitmap> = emptyList()) : BookLoadResult
    data class Error(val message: String) : BookLoadResult
}

class BookTextLoader(private val context: Context) {

    suspend fun loadBook(filePath: String, format: String): BookLoadResult = withContext(Dispatchers.IO) {
        if (filePath.isBlank()) return@withContext BookLoadResult.Error("Chemin de fichier invalide.")
        val uri = Uri.parse(filePath)

        runCatching {
            when (format.uppercase()) {
                "TXT" -> {
                    val text = loadTxt(uri)
                    val pages = chunkText(text, 1500)
                    BookLoadResult.Success(pages)
                }
                "EPUB" -> {
                    val chapters = loadEpubChapters(uri)
                    BookLoadResult.Success(chapters)
                }
                "PDF" -> {
                    loadPdfPages(uri)
                }
                else -> BookLoadResult.Error("Format non supporté pour la lecture directe.")
            }
        }.getOrElse { BookLoadResult.Error("Erreur de chargement : ${it.localizedMessage}") }
    }

    private fun loadTxt(uri: Uri): String {
        context.contentResolver.openInputStream(uri)?.use { inputStream ->
            BufferedReader(InputStreamReader(inputStream, Charsets.UTF_8)).use { reader ->
                return reader.readText()
            }
        }
        return "Impossible d'ouvrir le fichier texte."
    }

    private fun loadEpubChapters(uri: Uri): List<String> {
        val chapters = mutableListOf<String>()
        context.contentResolver.openInputStream(uri)?.use { inputStream ->
            ZipInputStream(inputStream).use { zip ->
                var entry = zip.nextEntry
                while (entry != null) {
                    val name = entry.name.lowercase()
                    if ((name.endsWith(".html") || name.endsWith(".xhtml") || name.endsWith(".htm")) &&
                        !name.contains("toc") && !name.contains("cover") && !name.contains("title")) {
                        
                        val content = zip.bufferedReader(Charsets.UTF_8).readText()
                        val cleanText = content
                            .replace(Regex("(?is)<style.*?>.*?</style>"), "")
                            .replace(Regex("(?is)<script.*?>.*?</script>"), "")
                            .replace(Regex("(?is)<head.*?>.*?</head>"), "")
                            .replace(Regex("<[^>]*>"), " ")
                            .replace(Regex("&nbsp;"), " ")
                            .replace(Regex("&amp;"), "&")
                            .replace(Regex("&quot;"), "\"")
                            .replace(Regex("[ \\t]+"), " ")
                            .replace(Regex("\n\\s*\n"), "\n\n")
                            .trim()

                        if (cleanText.length > 30) {
                            chapters.add(cleanText)
                        }
                    }
                    zip.closeEntry()
                    entry = zip.nextEntry
                }
            }
        }
        return if (chapters.isEmpty()) listOf("Aucun contenu textuel lisible trouvé dans cet EPUB.") else chapters
    }

    private fun loadPdfPages(uri: Uri): BookLoadResult {
        val descriptor: ParcelFileDescriptor = context.contentResolver.openFileDescriptor(uri, "r")
            ?: return BookLoadResult.Error("Impossible d'ouvrir le descripteur PDF.")
        
        val renderer = PdfRenderer(descriptor)
        val pagesList = mutableListOf<String>()
        
        for (i in 0 until renderer.pageCount) {
            pagesList.add("Page ${i + 1} sur ${renderer.pageCount} (Document PDF)")
        }
        renderer.close()
        descriptor.close()
        
        return BookLoadResult.Success(pagesList)
    }

    private fun chunkText(text: String, maxCharsPerPage: Int): List<String> {
        if (text.length <= maxCharsPerPage) return listOf(text)
        val chunks = mutableListOf<String>()
        var start = 0
        while (start < text.length) {
            val end = minOf(start + maxCharsPerPage, text.length)
            chunks.add(text.substring(start, end).trim())
            start = end
        }
        return chunks
    }
}

/**
 * Barre de navigation flottante avec encoche centrale et bouton « + ».
 *
 * - Les couleurs viennent de MaterialTheme.colorScheme : elles suivent donc
 *   automatiquement le thème de l'app (clair / sombre / sépia / dynamic color).
 * - Le « + » ouvre le sélecteur de fichiers Android et renvoie les Uri choisis.
 *   Permission de lecture persistante prise automatiquement.
 */
@Composable
fun FloatingActionNavBar(
    selected: ReaderTab,
    onTabSelected: (ReaderTab) -> Unit,
    onFilesPicked: (List<Uri>) -> Unit,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val context = LocalContext.current

    val picker = rememberLauncherForActivityResult(
        ActivityResultContracts.OpenMultipleDocuments()
    ) { uris ->
        if (uris.isNotEmpty()) {
            uris.forEach { uri ->
                try {
                    context.contentResolver.takePersistableUriPermission(
                        uri, Intent.FLAG_GRANT_READ_URI_PERMISSION
                    )
                } catch (_: SecurityException) {
                    // Le provider n'autorise pas la permission persistante : on garde l'accès temporaire.
                }
            }
            onFilesPicked(uris)
        }
    }

    val barHeight = 72.dp
    val fabOverflow = 46.dp // partie du bouton qui dépasse au-dessus de la barre
    val barShape = remember {
        NotchedBarShape(corner = 28.dp, notchHalfWidth = 52.dp, notchDepth = 24.dp)
    }

    Box(
        modifier
            .fillMaxWidth()
            .navigationBarsPadding()
            .padding(horizontal = 16.dp, vertical = 12.dp)
            .height(barHeight + fabOverflow)
    ) {
        Surface(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth()
                .height(barHeight),
            shape = barShape,
            color = colors.surfaceContainerHigh,
            contentColor = colors.onSurfaceVariant,
            shadowElevation = 10.dp,
            border = BorderStroke(1.dp, colors.outlineVariant.copy(alpha = 0.5f)),
        ) {
            Row(
                Modifier
                    .fillMaxSize()
                    .padding(horizontal = 12.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                val tabs = ReaderTab.entries
                tabs.take(2).forEach { tab ->
                    NavTab(tab, tab == selected, { onTabSelected(tab) }, Modifier.weight(1f))
                }
                Spacer(Modifier.width(72.dp)) // place du « + »
                tabs.drop(2).forEach { tab ->
                    NavTab(tab, tab == selected, { onTabSelected(tab) }, Modifier.weight(1f))
                }
            }
        }

        CenterAddButton(
            onClick = { picker.launch(READER_MIME_TYPES) },
            modifier = Modifier.align(Alignment.TopCenter),
        )
    }
}

@Composable

                onClick = onClick,

    ) {
        Box(

                .clip(RoundedCornerShape(18.dp))
                .background(pill),
            contentAlignment = Alignment.Center,

                contentDescription = tab.label,
                tint = tint,
                modifier = Modifier.size(24.dp),

        }
    }

/** Rectangle arrondi avec une encoche lisse (courbes de Bézier) au centre du bord haut. */
private class NotchedBarShape(
    private val corner: Dp,
    private val notchHalfWidth: Dp,
    private val notchDepth: Dp,
) : Shape {
    override fun createOutline(
        size: Size,
        layoutDirection: LayoutDirection,
        density: Density,
    ): Outline {
        val r = with(density) { corner.toPx() }
        val hw = with(density) { notchHalfWidth.toPx() }
        val d = with(density) { notchDepth.toPx() }
        val w = size.width
        val h = size.height
        val cx = w / 2f

        val path = Path().apply {
            moveTo(r, 0f)
            lineTo(cx - hw, 0f)
            cubicTo(cx - hw * 0.6f, 0f, cx - hw * 0.4f, d, cx, d)
            cubicTo(cx + hw * 0.4f, d, cx + hw * 0.6f, 0f, cx + hw, 0f)
            lineTo(w - r, 0f)
            arcTo(Rect(w - 2 * r, 0f, w, 2 * r), -90f, 90f, false)
            lineTo(w, h - r)
            arcTo(Rect(w - 2 * r, h - 2 * r, w, h), 0f, 90f, false)
            lineTo(r, h)
            arcTo(Rect(0f, h - 2 * r, 2 * r, h), 90f, 90f, false)
            lineTo(0f, r)
            arcTo(Rect(0f, 0f, 2 * r, 2 * r), 180f, 90f, false)
            close()
        }
        return Outline.Generic(path)
    }
}
